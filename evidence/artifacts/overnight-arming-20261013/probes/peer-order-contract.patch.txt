+        return importlib.import_module("overnight").check_overnight_order(payload)

+    def refused(self,payload):

+            self.guard(payload)
+        self.assertEqual(str(caught.exception),"overnight_order_contract_refused")

+    def test_limit_day_and_gtc_are_the_only_extended_order_shapes(self):
+        for side in ("buy","sell"):

+                    payload=order(side=side,time_in_force=tif)
+                    before=dict(payload)
+                    self.guard(payload)
+                    self.assertEqual(payload,before)
+
+    def test_non_object_payloads_are_refused_with_the_contract_reason(self):
+        for payload in (None,[],(),"limit",True):
+            with self.subTest(payload=payload):

+    def test_every_non_limit_type_is_refused_including_broker_trailing_stop(self):
