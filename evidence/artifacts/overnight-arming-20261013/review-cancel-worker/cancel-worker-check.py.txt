        source = subprocess.check_output(['git', 'show', '8f64d5a49:blueprints/us-equities/adaptive-paper/transport.py'], cwd=ROOT).decode()

        source = (ROOT / 'blueprints/us-equities/adaptive-paper/transport.py').read_text()
